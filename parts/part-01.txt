function tone(freq,dur,type,vol,slide){
  if(!actx) return;
  try{
    var o=actx.createOscillator(), g=actx.createGain();
    o.type=type;
    o.frequency.setValueAtTime(freq, actx.currentTime);
    if(slide) o.frequency.exponentialRampToValueAtTime(Math.max(40,slide), actx.currentTime+dur);
    g.gain.setValueAtTime(vol, actx.currentTime);
    g.gain.exponentialRampToValueAtTime(0.001, actx.currentTime+dur);
    o.connect(g); g.connect(actx.destination);
    o.start(); o.stop(actx.currentTime+dur);
  }catch(e){}
}
function unlock(){
  game.audio=1;
  var AC=window.AudioContext||window.webkitAudioContext;
  if(!AC) return;
  if(!actx) actx=new AC();
  if(actx.state==='suspended') actx.resume();
}
function burst(x,y,n,ci,spd){
  var left=n;
  for(var i=0;i<SP_N && left>0;i++){
    var s=SP[i];
    if(s.life>0) continue;
    var a=(i*2.399+game.frame)*1.7;
    var sp=spd*(0.45+(i%5)*0.18);
    s.x=x; s.y=y;
    s.vx=Math.cos(a)*sp;
    s.vy=Math.sin(a)*sp-0.8;
    s.life=12+(i%7);
    s.max=s.life;
    s.ci=ci;
    s.sz=2+(i&1);
    left--;
  }
}
function updateSparks(){
  for(var i=0;i<SP_N;i++){
    var s=SP[i];
    if(s.life<=0) continue;
    s.x+=s.vx; s.y+=s.vy; s.vy+=0.16; s.life--;
  }
}
function syncSec(){
  var s=Math.ceil(game.timeLeft/60);
  if(s<0) s=0;
  if(s!==game.secShown){ game.secShown=s; game.secStr=String(s); }
}
function cyclePick(slot,dir){
  var n=rosterCount();
  var i=game.picks[slot], other=game.picks[1-slot];
  var k;
  for(k=0;k<n;k++){
    i=(i+dir+n)%n;
    if(i!==other){ game.picks[slot]=i; return; }
  }
}

function tryStart(){
  game.boss=0;
  var n=rosterCount();
  if(game.picks[0]>=n) game.picks[0]=0;
  if(game.picks[1]>=n) game.picks[1]=n>1?1:0;
  if(game.picks[0]===game.picks[1]) cyclePick(1,1);
  tone(520,0.06,'square',0.04,360);
  startMatch();
}
function startBoss(){
  var bi=-1, i;
  for(i=0;i<ROSTER.length;i++) if(ROSTER[i].name==='THE BOSS') bi=i;
  if(bi<0) return;
  var n=rosterCount();
  if(game.picks[0]<0||game.picks[0]>=n) game.picks[0]=0;
  game.picks[1]=bi;
  game.versus=0;
  game.quest=0;
  game.boss=1;
  game.paused=0;
  tone(160,0.1,'sawtooth',0.05,48);
  startMatch();
}
function noteMatch(winner){
  if(game.versus) return 0;
  if(game.boss || (F[1].def && F[1].def.name==='THE BOSS')) return 0;
  if(!F[1].isCpu || F[1].def.id!==0) return 0;
  if(winner!==0){
    game.gojoStreak=0;
    lsSet('crosscut_gojo_streak', 0);
    return 0;
  }
  if(game.diff!==2) return 0;
  game.gojoStreak=(game.gojoStreak||0)+1;
  lsSet('crosscut_gojo_streak', game.gojoStreak);
  if(game.gojoStreak>=2 && !game.cooper){
    game.cooper=1;
    lsSet('crosscut_cooper', 1);
    return 1;
  }
  return 0;
}
function updateMenu(){
  game.versus=0;
  game.boss=0;
  var nPlay=rosterCount();
  if(game.picks[0]>=nPlay) game.picks[0]=0;
  if(game.picks[1]>=nPlay) game.picks[1]=game.picks[0]===0?1:0;
  if(edge.Digit3) cycleDiff();
  if(edge.KeyA) cyclePick(0,-1);
  if(edge.KeyD) cyclePick(0,1);
  if(edge.KeyW) cyclePick(1,-1);
  if(edge.KeyS) cyclePick(1,1);
  if(edge.Enter||edge.NumpadEnter) tryStart();
  if(mdown && hitBox(BOSS_BOX)){
    startBoss();
    return;
  } else if(mdown && game.cooper && hitBox(QUEST_BOX)){
    game.mode='questask';
    game.quest=0;
    game.paused=0;
    tone(500,0.05,'square',0.035,380);
  } else if(mdown){
    var dh=diffHit(mx,my);
    if(dh>=0) setDiff(dh);
    var nCard=rosterCount();
    var cpuPick=-1, youPick=-1, ci;
    for(ci=0;ci<nCard;ci++){
      var cb=cardBox(ci);
      var chip=cpuChipBox(cb);
      if(mx>=chip[0]&&mx<chip[0]+chip[2]&&my>=chip[1]&&my<chip[1]+chip[3]) cpuPick=ci;
      else if(mx>=cb.x && mx<=cb.x+cb.w && my>=cb.y && my<=cb.y+cb.h) youPick=ci;
    }
    if(hitBox(YOU_PREV)) cyclePick(0,-1);
    else if(hitBox(YOU_NEXT)) cyclePick(0,1);
    else if(hitBox(CPU_PREV)) cyclePick(1,-1);
    else if(hitBox(CPU_NEXT)) cyclePick(1,1);
    else if(cpuPick>=0){
      if(cpuPick!==game.picks[0]) game.picks[1]=cpuPick;
      else cyclePick(1,1);
    } else if(youPick>=0){
      game.picks[0]=youPick;
      if(game.picks[0]===game.picks[1]) cyclePick(1,1);
    }
    if(hitBox(START_BOX)) tryStart();
  }
  if(game.picks[0]===game.picks[1]) cyclePick(1,1);
  if(!game.cooper && (game.picks[0]>3||game.picks[1]>3)){
    game.picks[0]=0; game.picks[1]=1;
  }
}
function pushDir(f){
  var x=0;
  if(f.inL) x--;
  if(f.inR) x++;
  var forward=x===f.face;
  var back=x===-f.face && x!==0;
  var d=0;
  if(f.inD && forward) d=3;
  else if(f.inD && back) d=5;
  else if(f.inD) d=1;
  else if(forward) d=2;
  else if(back) d=4;
  f.dirs[f.dirI]=d;
  f.dirI=(f.dirI+1)%24;
}
function qcf(f){
  var st=0;
  for(var age=20; age>=0; age--){
    var d=f.dirs[(f.dirI-1-age+240)%24];
    if(st===0){ if(d===1||d===3) st=1; }
    else if(st===1){
      if(d===3) st=2;
      else if(d===2) st=3;
      else if(d===1) st=1;
      else if(d!==0 && d!==5) st=0;
    } else if(st===2){
      if(d===2||d===3) st=3;
      else if(d===1) st=1;
      else if(d!==0) st=0;
    } else if(st===3){
      if(d===2) st=3;
      else if(d===1) st=0;
    }
  }
  return st===3;
}

  f.inBlk=(held(map.blk)||(t&&touchHold.blk))?1:0;
  if(f.inL&&f.inR){ f.inL=0; f.inR=0; }
  if(edged(map.lp)||(t&&touchPulse.lp)) f.buf.lp=10;

  if(edged(map.hk)||(t&&touchPulse.hk)) f.buf.hk=10;

function begin(f,key){
  var mv=key==='spec'?SPEC[f.def.special]:MOVES[key];
  f.mv=mv; f.mvKey=key; f.kind=mv.kind||'';
  f.suMod=0;
  if(key==='spec') f.suMod=SPEC_WIND-mv.su;
  else if(f.def.lightSu && (key==='lp'||key==='lk'||key==='clp'||key==='clk'||key==='jlp')) f.suMod=f.def.lightSu;
  f.state=key==='spec'?'spec':'atk';
  f.timer=0; f.didHit=0; f.spawned=0;
  f.atkId=(f.atkId+1)&65535;
  f.atkCrouch=key==='clp'||key==='chp'||key==='clk'||key==='chk';
}
function tryAction(f){
  if(game.mode!=='fight') return;
  var st=f.state;
  if(st==='hit'||st==='ko'||st==='blockstun'||st==='atk'||st==='spec') return;
  var air=!f.grounded || st==='air';
  if(air){
    if(!f.airOk) return;
    if(f.buf.lp||f.buf.hp){ f.buf.lp=0; f.buf.hp=0; begin(f,'jlp'); }
    else if(f.buf.lk||f.buf.hk||f.buf.spec){ f.buf.lk=0; f.buf.hk=0; f.buf.spec=0; begin(f,'jhk'); }
    return;
  }
  if(f.buf.spec){ f.buf.spec=0; begin(f,'spec'); return; }
  if(f.buf.hp){ f.buf.hp=0; begin(f, f.inD?'chp':'hp'); return; }
  if(f.buf.hk){ f.buf.hk=0; begin(f, f.inD?'chk':'hk'); return; }
  if(f.buf.lp){ f.buf.lp=0; begin(f, f.inD?'clp':'lp'); return; }
  if(f.buf.lk){ f.buf.lk=0; begin(f, f.inD?'clk':'lk'); return; }
  if(f.inU && !f.inD){
    f.vy=f.def.jump; f.grounded=0; f.state='air'; f.airOk=1;
  }
}
function ai(cpu,foe){
  cpu.inL=cpu.inR=cpu.inU=cpu.inD=cpu.inBlk=0;
  var dx=foe.x-cpu.x;
  var ad=dx<0?-dx:dx;
  var dir=dx>=0?1:-1;
  if(!cpu.grounded){
    if(dx>8) cpu.inR=1; else if(dx<-8) cpu.inL=1;
    if(cpu.state==='air' && cpu.airOk && ad<150 && (game.frame%6)===0) cpu.buf.hk=8;
    return;
  }
  if(cpu.state==='ko'||cpu.state==='hit'||cpu.state==='atk'||cpu.state==='spec'||cpu.state==='blockstun') return;
  var pers=cpu.def.ai;
  var diff=game.diff;
  var r=(game.frame*17+cpu.def.id*13)%100;
  var blkChance=pers.block+(diff===2?16:diff===0?-12:0);
  if(blkChance<5) blkChance=5;
  if(blkChance>68) blkChance=68;
  var soonWindow=diff===2?14:diff===0?4:8;
  var activeSoon=0;
  if(foe.mv && (foe.state==='atk'||foe.state==='spec')){
    var su=foe.mv.su+foe.suMod;
    if(foe.timer>=su-soonWindow && foe.timer<su+foe.mv.ac && ad<(diff===2?210:185)) activeSoon=1;
  }
  if(activeSoon){
    if(cpu.react<0){
      if(diff===2) cpu.react=1;
      else if(diff===0) cpu.react=7+((game.frame+cpu.def.id)%4);
      else cpu.react=1+((game.frame+cpu.def.id)%3);
    }
  } else if(cpu.react>0) cpu.react=-1;
  if(cpu.react>0){
    cpu.react--;
    if(cpu.react===0 && activeSoon){
      var roll=(game.frame*5+cpu.def.id*17)%100;
      if(roll<blkChance && diff===0){
        cpu.blkT=8+(game.frame%4);
        cpu.crouchBlk=foe.mv && foe.mv.h===1?1:0;
        cpu.inBlk=1;
        if(cpu.crouchBlk) cpu.inD=1;
        return;
      }
      if(ad<160) cpu.buf.hp=8;
    }
  }
  if(cpu.blkT>0){
    cpu.blkT--;
    cpu.inBlk=1;
    if(cpu.crouchBlk) cpu.inD=1;
    return;
  }
  if(cpu.def.id===2){
    if(diff===0 && r>62){
      if(ad>48){ if(dir<0) cpu.inL=1; else cpu.inR=1; }
      return;
    }
    if(ad>64){
      if(dir<0) cpu.inL=1; else cpu.inR=1;
      if(ad<170) cpu.buf.spec=8;
      else if(r>50) cpu.inU=1;
      if(ad<110) cpu.buf.hp=8;
      return;
    }
    if(r<30) cpu.buf.spec=8;
    else if(r<52) cpu.buf.hp=8;
    else if(r<72) cpu.buf.hk=8;
    else if(r<86) cpu.buf.lp=8;
    else { cpu.inD=1; cpu.buf.hk=8; }
    return;
  }
  var foeSu=foe.mv?foe.mv.su+foe.suMod:0;
  var foeRec=foe.mv && (foe.state==='atk'||foe.state==='spec') && foe.timer>=foeSu+foe.mv.ac;
  if(foe.state==='hit' && ad<200){
    if(ad>78){ if(dir<0) cpu.inL=1; else cpu.inR=1; }
    if(ad<80){
      if(r<50) cpu.buf.lp=8;
      else if(r<78) cpu.buf.hp=8;
      else cpu.buf.lk=8;
    }
    return;
  }
  if(foeRec && ad<150){
    if(dir<0) cpu.inL=1; else cpu.inR=1;
    if(ad<82) cpu.buf.lp=8;
    return;
  }
  if(dir<0) cpu.inL=1; else cpu.inR=1;
  if(ad>150){
    if(r>40 && cpu.def.special!=='slam') cpu.buf.spec=8;
    if(ad<190 && r>55) cpu.inU=1;
    return;
  }
  if(foe.state==='hit' || foe.state==='blockstun'){
    if(r<40) cpu.buf.hp=8;
    else if(r<70) cpu.buf.hk=8;
    else cpu.buf.spec=8;
    return;
  }
  if(r<18) cpu.buf.lp=8;
  else if(r<40) cpu.buf.hp=8;
  else if(r<58) cpu.buf.hk=8;
  else if(r<72){ cpu.inD=1; cpu.buf.lk=8; }
  else if(r<88) cpu.buf.spec=8;
  else cpu.buf.lp=8;
}
function outDmg(raw,f){
  var d=Math.max(1, Math.round(raw*f.def.dmg));
  if(f.isCpu && game.diff===2) d=Math.max(1, Math.round(d*1.12));
  else if(f.isCpu && game.diff===0) d=Math.max(1, Math.round(d*0.8));
  return d;
}
function spawnProj(f,kind){
  var p=null;
  for(var i=0;i<PROJ_N;i++) if(!PROJ[i].on){ p=PROJ[i]; break; }
  if(!p) return;
  p.on=1; p.owner=f.slot; p.atkId=f.atkId;
  p.dmg=outDmg(f.mv.dmg, f);
  p.hs=f.mv.hs+(f.def.specStun||0);
  p.bs=f.mv.bs+(f.def.specBlock||0);
  p.push=f.mv.push; p.knock=f.mv.knock; p.shake=f.mv.shake; p.hgt=0; p.stop=f.mv.stop; p.vy=0;
  if(kind==='bolt'){
    p.kind=0; p.w=36; p.h=20; p.life=66; p.y=f.y-64; p.vx=f.face*7.3;
    p.x=f.face===1?f.x+28:f.x-28-p.w;
  } else if(kind==='shard'){
    p.kind=1; p.w=32; p.h=18; p.life=88; p.y=f.y-58; p.vx=f.face*5.2;
    p.x=f.face===1?f.x+22:f.x-22-p.w;
    p.hs+=4; p.bs+=3;
  } else if(kind==='glory'){
    p.kind=3; p.w=46; p.h=32; p.life=120; p.y=f.y-84; p.vx=f.face*10;
    p.x=f.face===1?f.x+30:f.x-30-p.w;
  } else {
    p.kind=2; p.w=34; p.h=34; p.life=42; p.y=f.y-72; p.vx=f.face*6.1;
    p.x=f.face===1?f.x+24:f.x-24-p.w;
  }
}
function updateFighter(f){
  if(f.x!==f.x||f.y!==f.y){ f.x=f.slot?700:260; f.y=GROUND; f.vx=0; f.vy=0; f.grounded=1; }
  if(game.mode==='fight') tryAction(f);
  decayBuf(f);
  if(f.state==='atk'||f.state==='spec'){
    if(f.state==='spec') f.buf.spec=0;
    f.timer++;
    var su=f.mv.su+f.suMod;
    var end=su+f.mv.ac+f.mv.re;
    if(f.timer>=end){
      f.mv=null; f.kind=''; f.mvKey='';
      f.state=f.grounded?'idle':'air';
