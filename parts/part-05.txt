function syncChrome(){
  var ui=document.getElementById('touchui');
  var fs=document.getElementById('fsbtn');
  if(ui){
    var on=game.mode==='intro'||game.mode==='fight'||game.mode==='ko';
    if(on) ui.removeAttribute('hidden');
    else ui.setAttribute('hidden','');
    var nodes=ui.querySelectorAll('button');
    for(var bi=0;bi<nodes.length;bi++){
      var act=nodes[bi].getAttribute('data-act');
      if(touchHold[act]) nodes[bi].classList.add('on');
      else nodes[bi].classList.remove('on');
    }
  }
  if(fs){
    var full=document.fullscreenElement||document.webkitFullscreenElement;
    var label=full?'EXIT':'FULLSCREEN';
    if(fs.textContent!==label) fs.textContent=label;
  }
}

function loop(ts){
  if(!lastT){ lastT=ts; requestAnimationFrame(loop); return; }
  var d=ts-lastT; lastT=ts;
  if(d>80) d=80;
  acc+=d;
  var n=0;
  while(acc>=16.6667 && n<4){ update(); acc-=16.6667; n++; }
  draw();
  requestAnimationFrame(loop);
}
var touchHold={l:0,r:0,u:0,d:0,lp:0,hp:0,lk:0,hk:0,blk:0,sp:0};
var touchPulse={lp:0,hp:0,lk:0,hk:0,sp:0};
var holdCount={l:0,r:0,u:0,d:0,lp:0,hp:0,lk:0,hk:0,blk:0,sp:0};
var fingerAct={};
var mouseAct='';
var eatMouse=0;
var fsStamp=0;
function pressAct(act){
  if(!act || act==='__fs__' || holdCount[act]===undefined) return;
  if(!holdCount[act] && touchPulse[act]!==undefined && game.mode==='fight') touchPulse[act]=1;
  holdCount[act]++;
  touchHold[act]=1;
}
function releaseAct(act){
  if(!act || act==='__fs__' || !holdCount[act]) return;
  holdCount[act]--;
  if(holdCount[act]<=0){ holdCount[act]=0; touchHold[act]=0; }
}
function actFromEl(el){
  while(el && el!==document.documentElement){
    if(el.getAttribute && el.getAttribute('data-act')) return el.getAttribute('data-act');
    el=el.parentNode;
  }
  return '';
}
function isFsEl(el){
  while(el && el!==document.documentElement){
    if(el.id==='fsbtn') return 1;
    el=el.parentNode;
  }
  return 0;
}
function canvasPoint(cx,cy){
  var canvas=document.getElementById('c');
  if(!canvas) return;
  var r=canvas.getBoundingClientRect();
  if(r.width<=0||r.height<=0) return;
  mx=(cx-r.left)*(W/r.width);
  my=(cy-r.top)*(H/r.height);
}
function toggleFs(){
  var full=document.fullscreenElement||document.webkitFullscreenElement;
  if(full){
    var ex=document.exitFullscreen||document.webkitExitFullscreen||document.webkitCancelFullScreen;
    if(ex){ try{ ex.call(document); }catch(err){} }
  } else {
    var root=document.documentElement;
    var req=root.requestFullscreen||root.webkitRequestFullscreen||root.msRequestFullscreen;
    if(req){ try{ req.call(root); }catch(err){} }
  }
}
function onFsEvent(e){
  if(e){ e.preventDefault(); e.stopPropagation(); }
  var now=Date.now();
  if(now-fsStamp<1000) return;
  fsStamp=now;
  unlock();
  toggleFs();
}
function onTouchStart(e){
  eatMouse=Date.now()+1000;
  if(e.cancelable) e.preventDefault();
  unlock();
  var i,t,el,act;
  for(i=0;i<e.changedTouches.length;i++){
    t=e.changedTouches[i];
    el=document.elementFromPoint(t.clientX, t.clientY);
    if(isFsEl(el)){ fingerAct[t.identifier]='__fs__'; continue; }
    act=actFromEl(el);
    if(act){
      fingerAct[t.identifier]=act;
      pressAct(act);
    } else {
      fingerAct[t.identifier]='';
      canvasPoint(t.clientX, t.clientY);
      mdown=1;
    }
  }
}
function onTouchMove(e){
  if(e.cancelable) e.preventDefault();
  var i,t,el,act,prev;
  for(i=0;i<e.changedTouches.length;i++){
    t=e.changedTouches[i];
    if(fingerAct[t.identifier]===undefined) continue;
    prev=fingerAct[t.identifier];
    if(prev==='__fs__') continue;
    el=document.elementFromPoint(t.clientX, t.clientY);
    act=actFromEl(el)||'';
    if(prev===act){
      if(!act) canvasPoint(t.clientX, t.clientY);
      continue;
    }
    if(prev) releaseAct(prev);
    fingerAct[t.identifier]=act;
    if(act) pressAct(act);
    else canvasPoint(t.clientX, t.clientY);
  }
}
function onTouchEnd(e){
  if(e.cancelable) e.preventDefault();
  var i,t,prev;
  for(i=0;i<e.changedTouches.length;i++){
    t=e.changedTouches[i];
    prev=fingerAct[t.identifier];
    if(prev==='__fs__') onFsEvent(null);
    else if(prev) releaseAct(prev);
    delete fingerAct[t.identifier];
  }
}
function releaseAllTouch(){
  var k;
  fingerAct={};
  mouseAct='';
  for(k in holdCount){ holdCount[k]=0; touchHold[k]=0; }
}
function onDown(e){
  if(e.repeat) return;
  var c=e.code;
  if(down[c]===undefined) return;
  if(!down[c]) edge[c]=1;
  down[c]=1;
  if(c==='ArrowLeft'||c==='ArrowRight'||c==='ArrowUp'||c==='ArrowDown'||c==='Space') e.preventDefault();
  unlock();
}
function onUp(e){ if(down[e.code]!==undefined) down[e.code]=0; }
function boot(){
  var canvas=document.getElementById('c');
  canvas.tabIndex=1;
  ctx=canvas.getContext('2d',{alpha:false});
  window.addEventListener('keydown', onDown);
  window.addEventListener('keyup', onUp);
  canvas.addEventListener('mousedown', function(e){
    if(Date.now()<eatMouse) return;
    unlock();
    canvas.focus();
    var r=canvas.getBoundingClientRect();
    if(r.width>0){
      mx=(e.clientX-r.left)*(W/r.width);
      my=(e.clientY-r.top)*(H/r.height);
      mdown=1;
    }
  });
  canvas.addEventListener('contextmenu', function(e){ e.preventDefault(); });
  var opt={capture:true, passive:false};
  document.addEventListener('touchstart', onTouchStart, opt);
  document.addEventListener('touchmove', onTouchMove, opt);
  document.addEventListener('touchend', onTouchEnd, opt);
  document.addEventListener('touchcancel', onTouchEnd, opt);
  document.addEventListener('gesturestart', function(e){ e.preventDefault(); }, opt);
  document.addEventListener('gesturechange', function(e){ e.preventDefault(); }, opt);
  document.addEventListener('visibilitychange', releaseAllTouch);
  var fs=document.getElementById('fsbtn');
  if(fs){
    fs.addEventListener('click', onFsEvent);
    fs.addEventListener('touchend', onFsEvent, {passive:false});
  }
  var nodes=document.querySelectorAll('#touchui button');
  var bi;
  for(bi=0;bi<nodes.length;bi++){
    nodes[bi].addEventListener('mousedown', function(e){
      if(Date.now()<eatMouse){ e.preventDefault(); return; }
      e.preventDefault();
      e.stopPropagation();
      unlock();
      var act=e.currentTarget.getAttribute('data-act');
      if(mouseAct && mouseAct!==act) releaseAct(mouseAct);
      if(mouseAct!==act){ mouseAct=act; pressAct(act); }
    });
  }
  window.addEventListener('mouseup', function(){
    if(!mouseAct) return;
    releaseAct(mouseAct);
    mouseAct='';
  });
  requestAnimationFrame(loop);
}
