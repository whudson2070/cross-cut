        game.roundIndex++;
        startRound();
      }
    }
  }
  if(game.shake>0){ game.shake*=0.82; if(game.shake<0.35) game.shake=0; }
  updateSparks();
  clearEdges();
}

function poseOf(f){
  if(f.state==='ko') return 7;
  if(f.state==='hit') return 6;
  if(f.state==='block'||f.state==='cblock'||f.state==='blockstun') return 5;
  if(f.state==='spec'){
    if(f.mv && f.timer<f.mv.su+f.suMod) return 11;
    return 8;
  }
  if(f.state==='atk'){
    var k=f.mvKey;
    if(k==='lk'||k==='hk'||k==='clk'||k==='chk'||k==='jhk') return 4;
    return 3;
  }
  if(!f.grounded||f.state==='air') return 9;
  if(f.victory&&(f.state==='idle'||f.state==='walk')) return 10;
  if(f.state==='crouch') return 2;
  if(f.state==='walk') return 1;
  return 0;
}
