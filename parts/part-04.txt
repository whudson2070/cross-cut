function drawFighter(f){
  ctx.save();
  ctx.translate(f.x|0, f.y|0);
  ctx.scale(f.face<0?-1:1, 1);

  ctx.beginPath(); ctx.ellipse(0,-2,f.def.rad+8,7,0,0,6.2832); ctx.fill();

    var ang=Math.min(1.15, f.timer*0.05);
    ctx.rotate(-ang);
  }
  var bob=(f.state==='idle'||f.state==='walk'||f.state==='block'||f.state==='cblock')?Math.sin(game.frame*0.12+f.def.id)*1.5:0;
  ctx.translate(0,bob);
  tint=(f.flash>1&&(game.blink&1))?'#fff':'';

  else if(f.def.id===1) drawIce(f);
  else if(f.def.id===2) drawRed(f);
  else if(f.def.id===3) drawWill(f);
  else if(f.def.id===5) drawBoss(f);
  else drawCooper(f);
  if(f.state==='spec' && f.mv && f.timer<f.mv.su+f.suMod){

    ctx.stroke();
    ctx.globalAlpha=1;
  }
  tint='';
  ctx.restore();
}
